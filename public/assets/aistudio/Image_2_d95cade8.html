3export function useChatStorage(key: string, initialData: T) {
4  const [data, setData] = useState(() => {
5    if (typeof window === 'undefined') return initialData;
6    try {
7      const stored = window.localStorage.getItem(key);
8      return stored ? JSON.parse(stored) : initialData;
9    } catch (err) {
10      console.warn('Failed to load chat history', err);
11      return initialData;
12    }
13  });
14
15  const save = (newData: T) => {
16    setData(newData);
17    try {
18      window.localStorage.setItem(key, JSON.stringify(newData));
19    } catch (e: any) {
20      if (e.name === 'QuotaExceededError') {
21        // Truncate earliest records to restore capacity
22        const truncated = Array.isArray(newData) ? newData.slice(-20) : newData;
23        window.localStorage.setItem(key, JSON.stringify(truncated));
24      }
25    }
26  };
27
28  return [data, save] as const;
